    function getYouTubeId(url) {
      try {
        const u = new URL(url);
        if (u.hostname.includes("youtu.be")) return u.pathname.slice(1);
        if (u.pathname.startsWith("/shorts/")) return u.pathname.split("/")[2];
        if (u.pathname.startsWith("/embed/")) return u.pathname.split("/")[2];
        const v = u.searchParams.get("v");
        if (v) return v;
      } catch (_) { }
      const m = String(url).match(/(?:(?:v=)|\/)([0-9A-Za-z_-]{11})(?:[?&/]|\b)/);
      return m ? m[1] : null;
    }

    async function loadNotifications() {
      const notificationsList = document.getElementById("notificationsList");
      const noNotifications = document.getElementById("noNotifications");
      if (!notificationsList) return;

      try {
        const snap = await getDoc(doc(db, "notificacoes", "global"));
        if (!snap.exists() || !snap.data().active) {
          notificationsList.innerHTML = "";
          if (noNotifications) noNotifications.style.display = "block";
          return;
        }
        if (noNotifications) noNotifications.style.display = "none";
        notificationsList.innerHTML = "";
        const n = snap.data();
        const item = document.createElement("div");
        item.className = "notification-item";
        item.innerHTML = `
          <div class="notification-content">
            <p>${n.message || ""}</p>
            <small>${n.sender || ""} - ${n.ts?.seconds ? new Date(n.ts.seconds * 1000).toLocaleString("pt-BR") : ""}</small>
          </div>
        `;
        notificationsList.appendChild(item);
      } catch (err) {
        console.error("Erro ao carregar notificações:", err);
      }
